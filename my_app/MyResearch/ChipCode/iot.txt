#include <Wire.h>
#include "MAX30105.h"
#include "heartRate.h"

// -----------------------
// OLED Libraries
// -----------------------
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET -1
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);

// -----------------------
// MAX30102 Sensor
// -----------------------
MAX30105 particleSensor;

// Heart rate tracking
const byte RATE_SIZE = 4;
byte rates[RATE_SIZE];
byte rateSpot = 0;
long lastBeat = 0;
float beatsPerMinute = 0;
int beatAvg = 0;

// graph data buffer
#define GRAPH_WIDTH 128
int graphData[GRAPH_WIDTH] = {0};

unsigned long lastSendTime = 0;
const unsigned long sendInterval = 1000;

unsigned long lastBeatDetected = 0;
unsigned long zeroStartTime = 0;
bool zeroStateActive = false;
bool zeroSent = false;

// -----------------------
// DRAW GRAPH FUNCTION
// -----------------------
void drawGraph() {
  int maxValue = 150;
  int minValue = 40;

  // ----- Draw axes -----
  // Y-axis (left side)
  display.drawLine(0, 30, 0, 63, SSD1306_WHITE);

  // X-axis (bottom)
  display.drawLine(0, 63, 127, 63, SSD1306_WHITE);

  // ----- Draw graph line -----
  for (int i = 1; i < GRAPH_WIDTH; i++) {
    int x1 = i - 1;
    int y1 = map(graphData[i - 1], minValue, maxValue, 63, 30);
    int x2 = i;
    int y2 = map(graphData[i], minValue, maxValue, 63, 30);

    display.drawLine(x1, y1, x2, y2, SSD1306_WHITE);
  }
}


// -----------------------
// SETUP
// -----------------------
void setup() {
  Serial.begin(115200);
  Serial.println("Initializing...");

  Wire.begin(21, 22);  // SAME I2C bus for MAX30102 + OLED

  // ---- OLED INIT ----
  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println(" OLED not found!");
    while (1);
  }
  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0, 0);
  display.println("OLED Ready...");
  display.display();
  delay(800);

  // ---- MAX30102 INIT ----
  if (!particleSensor.begin(Wire, I2C_SPEED_FAST)) {
    Serial.println(" MAX30102 not found. Check wiring.");
    while (1);
  }

  particleSensor.setup();
  particleSensor.setPulseAmplitudeRed(0x0A);
  particleSensor.setPulseAmplitudeGreen(0);

  Serial.println(" MAX30102 initialized.");
}

// -----------------------
// LOOP
// -----------------------
void loop() {
  long irValue = particleSensor.getIR();
  bool beatDetected = false;

  // Heartbeat Detection
  if (checkForBeat(irValue)) {
    long delta = millis() - lastBeat;
    lastBeat = millis();

    beatsPerMinute = 60 / (delta / 1000.0);
    if (beatsPerMinute < 255 && beatsPerMinute > 20) {
      rates[rateSpot++] = (byte)beatsPerMinute;
      rateSpot %= RATE_SIZE;

      beatAvg = 0;
      for (byte x = 0; x < RATE_SIZE; x++) {
        beatAvg += rates[x];
      }
      beatAvg /= RATE_SIZE;

      beatDetected = true;
      lastBeatDetected = millis();

      zeroStartTime = 0;
      zeroStateActive = false;
      zeroSent = false;
    }
  }

  // Reset if no beat for 10 sec
  if (!beatDetected && (millis() - lastBeatDetected > 10000)) {
    beatsPerMinute = 0;
    beatAvg = 0;
  }

  // -----------------------
  // UPDATE GRAPH BUFFER
  // -----------------------
  for (int i = 1; i < GRAPH_WIDTH; i++) {
    graphData[i - 1] = graphData[i];
  }
  graphData[GRAPH_WIDTH - 1] = beatAvg;  // add new BPM point

  // -----------------------
  // OLED DISPLAY OUTPUT
  // -----------------------
  display.clearDisplay();

  // Title
  display.setTextSize(1);
  display.setCursor(0, 0);
  display.println("Heart Rate Monitor");

  // AVG BPM
  display.setTextSize(2);
  display.setCursor(0, 15);
  display.print("AVG:");
  display.println(beatAvg);

  // Graph
  drawGraph();

  display.display();

  // -----------------------
  // SERIAL OUTPUT
  // -----------------------
  Serial.print("BPM = ");
  Serial.print(beatsPerMinute, 2);
  Serial.print(" | Avg BPM = ");
  Serial.println(beatAvg);

  // Finger detection logic
  if (irValue < 50000) {
    Serial.println(" No finger detected.");
    beatsPerMinute = 0;
    beatAvg = 0;

    if (!zeroStateActive) {
      zeroStartTime = millis();
      zeroStateActive = true;
    }

    if (!zeroSent && millis() - zeroStartTime >= 3000) {
      Serial.println("➡ Sending zero values (no finger detected)");
      Serial.println("BPM: 0 | Avg BPM: 0");
      zeroSent = true;
      lastSendTime = millis();
    }
  } else {
    zeroStartTime = 0;
    zeroStateActive = false;
    zeroSent = false;

    if (beatDetected && millis() - lastSendTime >= sendInterval) {
      Serial.print(" Sending values | BPM: ");
      Serial.print(beatsPerMinute);
      Serial.print(" | Avg BPM: ");
      Serial.println(beatAvg);

      lastSendTime = millis();
    }
  }
}
